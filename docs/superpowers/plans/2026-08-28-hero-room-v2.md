`src/components/landing/HeroLamp/ideaTexture.ts`:
```ts
import { CanvasTexture, SRGBColorSpace } from "three";

/** The laptop's screen: the visitor's idea, set in Lora, cream on midnight.
 *  Redrawn only when the text changes — one 2D paint per keystroke. */
export function makeIdeaTexture(midnight: string, cream: string) {
  const W = 1024, H = 700, PAD = 72;
  const canvas = document.createElement("canvas");
  canvas.width = W; canvas.height = H;
  const ctx = canvas.getContext("2d")!;
  const texture = new CanvasTexture(canvas);
  texture.colorSpace = SRGBColorSpace;
  texture.flipY = false; // her screen plane is rotated π about X

  function wrap(text: string, max: number): string[] {
    const words = text.split(/\s+/).filter(Boolean);
    const lines: string[] = []; let line = "";
    for (const w of words) {
      const t = line ? `${line} ${w}` : w;
      if (ctx.measureText(t).width > max && line) { lines.push(line); line = w; } else line = t;
    }
    if (line) lines.push(line);
    return lines.slice(-6); // the last six lines fit the screen
  }

  function draw(text: string) {
    ctx.fillStyle = midnight; ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = cream;
    ctx.font = "500 56px Lora, Georgia, serif";
    ctx.textBaseline = "top";
    const lines = text ? wrap(text, W - PAD * 2) : [];
    let y = PAD;
    for (const l of lines) { ctx.fillText(l, PAD, y); y += 72; }
    // caret after the last line (or at the start when empty)
    const last = lines[lines.length - 1] ?? "";
    const cx = PAD + (last ? ctx.measureText(last).width + 8 : 0);
    const cy = lines.length ? y - 72 : PAD;
    ctx.fillRect(cx, cy + 4, 4, 60);
    texture.needsUpdate = true;
  }
  draw("");
  return { texture, draw };
}
```
Note: `document.fonts.ready.then(() => draw(getIdeaText()))` is called once from `Laptop` so the first paint after Lora loads uses Lora.

import * as THREE from "three";
import { useFrame } from "@react-three/fiber";
import { useGLTF, useTexture } from "@react-three/drei";
import { RectAreaLightUniformsLib } from "three/examples/jsm/lights/RectAreaLightUniformsLib.js";
import { makeIdeaTexture } from "./ideaTexture";

import { easeStrike, type LampColors } from "./Lamp";
import { DESK_TOP_Y } from "./Desk";

const SCALE = 0.055;

let rectInit = false;

export function Laptop({
  litRef, appearRef, colors, shadows,
}: { litRef: React.MutableRefObject<number>; appearRef: React.MutableRefObject<number>; colors: LampColors; shadows: boolean }) {
  const { scene: model } = useGLTF(LAPTOP_URL);
  const keysTex = useTexture(KEYS_URL);
  const root = React.useRef<THREE.Group>(null);
  const lid = React.useRef<THREE.Group>(null);
  const screenMat = React.useRef<THREE.MeshBasicMaterial>(null);
  const light = React.useRef<THREE.RectAreaLight>(null);

  // Owner's exception: her materials, her hex. The ONLY hex in the scene.
  const mats = React.useMemo(() => ({
    metal: new THREE.MeshStandardMaterial({ color: 0xcecfd3 }),
    dark: new THREE.MeshStandardMaterial({ color: 0x000000, roughness: 0.9, metalness: 0.9 }),
    cam: new THREE.MeshBasicMaterial({ color: 0x333333 }),
    logo: new THREE.MeshBasicMaterial({ color: 0xffffff }),
  }), []);
  const idea = React.useMemo(() => makeIdeaTexture(colors.midnight, colors.cream), [colors.midnight, colors.cream]);

  // Split her model into lid (_top) and base (_bottom) groups, assign by name.
  const parts = React.useMemo(() => {
    const top = model.getObjectByName("_top") as THREE.Group;
    const bottom = model.getObjectByName("_bottom") as THREE.Group;
    const byName: Record<string, THREE.Material> = {
      lid: mats.metal, base: mats.metal, logo: mats.logo, camera: mats.cam,
      "screen-frame": mats.dark, legs: mats.dark, keyboard: mats.dark, inner: mats.dark,
    };
    for (const g of [top, bottom]) g.traverse((o) => {
      if ((o as THREE.Mesh).isMesh) {
        const m = o as THREE.Mesh; m.material = byName[m.name] ?? mats.dark;
        m.castShadow = shadows; m.receiveShadow = m.name === "base";
      }

    if (!rectInit) { RectAreaLightUniformsLib.init(); rectInit = true; }

    const off = subscribe(redraw);
    document.fonts?.ready.then(redraw);
    return () => { off(); idea.texture.dispose(); Object.values(mats).forEach((m) => m.dispose()); };
  }, [idea, mats]);

  useFrame(() => {
    const k = easeStrike(litRef.current);
    const a = 1 - Math.pow(1 - appearRef.current, 3);
    // her laptopAppearTl: from lying flat below the frame to seated on the desk
    if (root.current) {
      root.current.rotation.x = 0.5 * Math.PI * (1 - a) + 0.05 * Math.PI * a;
      root.current.rotation.y = 0.2 * Math.PI * (1 - a) - 0.18 * a;
      root.current.position.y = DESK_TOP_Y - 2.2 * (1 - a);
    }
    // her laptopOpeningTl over the first 80% of the strike
    const o = Math.min(1, k / 0.8);
    if (lid.current) {
      lid.current.rotation.x = LID_CLOSED + (LID_OPEN - LID_CLOSED) * o;
      lid.current.position.z = 0.5 * (1 - o);
    }
    // her screenOnTl at 5% of the opening

    if (screenMat.current) screenMat.current.opacity = 0.96 * s;
    if (light.current) light.current.intensity = 1.5 * s;
  });

  return (
    <group ref={root} position={[0.55, DESK_TOP_Y, 0.4]} scale={SCALE}>
      <group ref={lid} rotation={[LID_CLOSED, 0, 0]}>
        <primitive object={parts.top} />
        {/* her screen: a plane in lid space, a dark plane 1mm behind for the off state */}
        <mesh position={[0, 10.5, -0.11]} rotation={[Math.PI, 0, 0]}>
          <planeGeometry args={SCREEN} />
          <meshBasicMaterial ref={screenMat} map={idea.texture} transparent opacity={0} side={THREE.BackSide} toneMapped={false} />
        </mesh>
        <mesh position={[0, 10.5, -0.111]} rotation={[Math.PI, Math.PI, 0]}>

          <primitive object={mats.dark} attach="material" />
        </mesh>
        <rectAreaLight ref={light} color={colors.cream} intensity={0} width={SCREEN[0]} height={SCREEN[1]} position={[0, 10.5, 0]} rotation={[Math.PI, 0, 0]} />
      </group>
